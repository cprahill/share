function terrainH(x, z) {
  return Math.sin(x * 0.045) * Math.cos(z * 0.038) * 0.55 + Math.sin(x * 0.11 + z * 0.07) * 0.18;
}

function placeOnTrack(t) {
  const p = curve.getPointAt(t);
  const tan = curve.getTangentAt(t);
  p.y = terrainH(p.x, p.z);
  return { p, tan };
}

function projectTrack(pos) {
  let bestT = 0, bestD = 1e9, bestP = null, bestTan = null;
  const n = 80;
  for (let i = 0; i < n; i++) {
    const t = i / n;
    const p = curve.getPointAt(t);
    const d = (p.x - pos.x) * (p.x - pos.x) + (p.z - pos.z) * (p.z - pos.z);
    if (d < bestD) { bestD = d; bestT = t; bestP = p; }
  }
  const refine = 12, span = 1 / n;
  for (let i = -refine; i <= refine; i++) {
    const t = (bestT + (i / refine) * span + 1) % 1;
    const p = curve.getPointAt(t);
    const d = (p.x - pos.x) * (p.x - pos.x) + (p.z - pos.z) * (p.z - pos.z);
    if (d < bestD) { bestD = d; bestT = t; bestP = p; }
  }
  bestTan = curve.getTangentAt(bestT);
  const right = new THREE.Vector3(bestTan.z, 0, -bestTan.x).normalize();
  const relx = pos.x - bestP.x;
  const relz = pos.z - bestP.z;
  const offset = relx * right.x + relz * right.z;
  return { t: bestT, p: bestP, tan: bestTan, right, offset };
}

function wallStrip(side) {
  const pos = [];
  const idx = [];
  const n = 140;
  for (let i = 0; i <= n; i++) {
    const t = i / n;
    const p = curve.getPointAt(t);
    const tan = curve.getTangentAt(t);
    const r = new THREE.Vector3(tan.z, 0, -tan.x).normalize();
    const h = 10 + Math.abs(Math.sin(t * 18.0)) * 14 + (side > 0 ? 4 : 0);
    const base = HALF_W + 1.2;
    const x0 = p.x + r.x * side * base;
    const z0 = p.z + r.z * side * base;
    const x1 = p.x + r.x * side * (base + 7);
    const z1 = p.z + r.z * side * (base + 7);
    const y0 = terrainH(x0, z0);
    pos.push(x0, y0, z0, x0, y0 + h, z0, x1, y0 + h * 0.72, z1, x1, y0 - 1, z1);
  }
  for (let i = 0; i < n; i++) {
    const o = i * 4;
    const p = o + 4;
    idx.push(o, o + 1, p + 1, o, p + 1, p);
    idx.push(o + 1, o + 2, p + 2, o + 1, p + 2, p + 1);
  }
  const geo = new THREE.BufferGeometry();
  geo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
  geo.setIndex(idx);
  geo.computeVertexNormals();
  const mesh = new THREE.Mesh(geo, rockMat(side > 0 ? ROCK_LIT : ROCK_MID, 0.92));
  mesh.castShadow = true;
  mesh.receiveShadow = true;
  scene.add(mesh);
}

function makeGate(index, t) {
  const { p, tan } = placeOnTrack(t);
  const right = new THREE.Vector3(tan.z, 0, -tan.x).normalize();
  const fwd = new THREE.Vector3(tan.x, 0, tan.z).normalize();
  const g = new THREE.Group();
  const y = terrainH(p.x, p.z);
  const hw = HALF_W - 1.2;
  const postGeo = new THREE.BoxGeometry(0.28, 6.2, 0.28);
  [-1, 1].forEach((s) => {
    const post = new THREE.Mesh(postGeo, trussMat);
    post.position.set(right.x * s * hw, y + 3.1, right.z * s * hw);
    post.castShadow = true;
    g.add(post);
  });
  const beam = new THREE.Mesh(new THREE.BoxGeometry(hw * 2 + 0.4, 0.28, 0.28), trussMat);
  beam.position.set(0, y + 6.2, 0);
  beam.lookAt(beam.position.clone().add(fwd));
  g.add(beam);
  beam.position.copy(p);
  beam.position.y = y + 6.2;
  beam.quaternion.setFromUnitVectors(new THREE.Vector3(1, 0, 0), right);
  const board = new THREE.Mesh(
    new THREE.BoxGeometry(4.6, 0.9, 0.12),
    new THREE.MeshStandardMaterial({ map: cpTex, roughness: 0.5, metalness: 0.1, emissive: 0x332200, emissiveIntensity: 0.35 })
  );
  board.position.set(p.x, y + 6.7, p.z);
  board.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, 1), fwd.clone().negate());
  g.add(board);
  [-1, 1].forEach((s) => {
    const flag = new THREE.Mesh(new THREE.BoxGeometry(0.08, 3.6, 0.7), bannerMat);
    flag.position.set(p.x + right.x * s * (hw + 0.6), y + 2.4, p.z + right.z * s * (hw + 0.6));
    flag.castShadow = true;
    g.add(flag);
    const xmark = new THREE.Mesh(new THREE.BoxGeometry(0.1, 0.7, 0.12), blackBar);
    xmark.position.copy(flag.position);
    xmark.position.y += 1.1;
    g.add(xmark);
  });
  scene.add(g);
  const glow = new THREE.Mesh(
    new THREE.BoxGeometry(hw * 2, 3.2, 0.4),
    new THREE.MeshStandardMaterial({ color: BANNER, transparent: true, opacity: 0.08, depthWrite: false })
  );
  glow.position.set(p.x, y + 1.6, p.z);
  glow.quaternion.setFromUnitVectors(new THREE.Vector3(0, 0, 1), fwd);
  g.add(glow);
  return { index, t, pos: p.clone().setY(y), forward: fwd, right, halfW: hw, mesh: g, glow };
}
